==================================================
CONFIGURACION DEL SERVIDOR WEB
UBUNTU SERVER 24.04.1
==================================================

DATOS DE RED

Interfaz:
ens3

IP:
10.21.40.130

Mascara:
255.255.255.240

Prefijo:
/28

Gateway:
10.21.40.129

DNS:
8.8.8.8
1.1.1.1


==================================================
1. EDITAR NETPLAN
==================================================

sudo nano /etc/netplan/50-cloud-init.yaml


CONTENIDO:

network:
  version: 2
  ethernets:
    ens3:
      dhcp4: false
      addresses:
        - 10.21.40.130/28
      routes:
        - to: default
          via: 10.21.40.129
      nameservers:
        addresses:
          - 8.8.8.8
          - 1.1.1.1


==================================================
2. VALIDAR NETPLAN
==================================================

sudo netplan generate


==================================================
3. APLICAR CONFIGURACION
==================================================

sudo netplan apply


==================================================
5. VERIFICAR RUTAS
==================================================

ip route


==================================================
6. PROBAR GATEWAY
==================================================

ping -c 4 10.21.40.129


==================================================
7. ACTUALIZAR SISTEMA
==================================================

sudo apt update


==================================================
8. INSTALAR APACHE2 Y OPENSSL
==================================================

sudo apt install apache2 openssl -y


==================================================
9. HABILITAR SSL EN APACHE
==================================================

sudo a2enmod ssl


==================================================
10. HABILITAR SITIO HTTPS
==================================================

sudo a2ensite default-ssl


==================================================
11. REINICIAR APACHE
==================================================

sudo systemctl restart apache2


==================================================
12. HABILITAR APACHE AL INICIO
==================================================

sudo systemctl enable apache2


==================================================
13. VERIFICAR APACHE
==================================================

sudo systemctl status apache2


==================================================
14. CREAR PAGINA WEB DEL LABORATORIO
==================================================

echo "<h1>Servidor Web - Laboratorio VPN FortiGate</h1><p>Conexion HTTPS exitosa a traves de la VPN Site-to-Site</p>" | sudo tee /var/www/html/index.html


==================================================
15. PERMITIR HTTPS EN EL FIREWALL
==================================================

sudo ufw allow 443/tcp


==================================================
16. PERMITIR HTTP OPCIONALMENTE
==================================================

sudo ufw allow 80/tcp


==================================================
17. VERIFICAR FIREWALL
==================================================

sudo ufw status


==================================================
18. VERIFICAR PUERTO 443
==================================================

sudo ss -tlnp | grep :443


==================================================
19. VERIFICAR PUERTO 80
==================================================

sudo ss -tlnp | grep :80


==================================================
20. PROBAR HTTPS LOCALMENTE
==================================================

curl -k https://localhost


O:

curl -k https://10.21.40.130


==================================================
21. URL FINAL DEL SERVIDOR
==================================================

https://10.21.40.130
